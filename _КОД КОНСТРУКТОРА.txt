function hexToSoft(hex, dark){
  try{
    const h = hex.replace('#','');
    const r=parseInt(h.substr(0,2),16), g=parseInt(h.substr(2,2),16), b=parseInt(h.substr(4,2),16);

}

function moveCard(id, dir){
  const sorted = [...DATA.cards].sort((a,b)=>(a.order||0)-(b.order||0));
  const i = sorted.findIndex(c=>c.id===id); if(i<0) return;
  const j = i+dir; if(j<0||j>=sorted.length) return;
  const a = sorted[i], b = sorted[j];
  const t = a.order||0; a.order = b.order||0; b.order = t;
  saveData(); adminCards();
}

function handleCardPhotos(e){
  const files = Array.from(e.target.files||[]);
  files.forEach(f=>{
    const reader = new FileReader();
    reader.onload = ev => { STATE._editingPhotos.push(ev.target.result); renderCardPhotoGrid(); };
    reader.readAsDataURL(f);
  });
  e.target.value='';
}
function saveCard(){
  const name = document.getElementById('cfName').value.trim();
  if(!name){ toast('Введите название','error'); return; }
  const obj = {
    id: STATE.editingCardId || uid(),
    categoryId: document.getElementById('cfCat').value,
    name,
    price: Number(document.getElementById('cfPrice').value)||0,
    oldPrice: Number(document.getElementById('cfOldPrice').value)||0,
    description: document.getElementById('cfDesc').value.trim(),
    photos: STATE._editingPhotos || [],
    tags: document.getElementById('cfTags').value.split(',').map(s=>s.trim()).filter(Boolean),
    available: document.getElementById('cfAvail').checked,
    order: STATE.editingCardId ? (DATA.cards.find(c=>c.id===STATE.editingCardId).order ?? DATA.cards.length) : DATA.cards.length,
    fields: {}
  };
  document.querySelectorAll('[data-cf]').forEach(inp=>{ obj.fields[inp.dataset.cf] = inp.value; });
  if(STATE.editingCardId){
    const i = DATA.cards.findIndex(c=>c.id===STATE.editingCardId);
    DATA.cards[i] = Object.assign(DATA.cards[i], obj);
  } else {
    DATA.cards.push(obj);
  }
  saveData();
  closeModal();
  adminCards();
  toast('Карточка сохранена','success');
}

function shadeColor(hex, percent){
  try{
    const h=hex.replace('#',''); let r=parseInt(h.substr(0,2),16),g=parseInt(h.substr(2,2),16),b=parseInt(h.substr(4,2),16);
    r=Math.max(0,Math.min(255,r+Math.round(r*percent/100)));
    g=Math.max(0,Math.min(255,g+Math.round(g*percent/100)));
    b=Math.max(0,Math.min(255,b+Math.round(b*percent/100)));
    return '#'+[r,g,b].map(x=>x.toString(16).padStart(2,'0')).join('');
  }catch(e){ return hex; }
}

function publishSite(){
  // Clone current document, replace embedded data, clear dynamic containers
  const clone = document.documentElement.cloneNode(true);
  // Write current data into the app-data script
  const dataScript = clone.querySelector('#app-data');
  if(dataScript) dataScript.textContent = JSON.stringify(DATA, null, 2);
  // Clear dynamic content so file stays clean
  ['navCategories','cardsGrid','filterPills','footerInner','adminContent','adminNav','modalRoot','toastRoot'].forEach(id=>{
    const el = clone.querySelector('#'+id); if(el) el.innerHTML='';
  });
  ['heroImg'].forEach(id=>{ const el=clone.querySelector('#'+id); if(el){ el.removeAttribute('src'); el.classList.add('hidden'); } });
  // Reset storefront visible
  const sf = clone.querySelector('#storefront'); if(sf) sf.classList.remove('hidden');
  const av = clone.querySelector('#adminView'); if(av) av.classList.add('hidden');
  // Build full HTML
  const html = '<!DOCTYPE html>\n' + clone.outerHTML;
  const blob = new Blob([html], {type:'text/html'});
  const a = document.createElement('a');
  const fname = (DATA.settings.siteTitle||'site').replace(/[^\w\u0400-\u04FFа-яА-Я-]+/g,'_').slice(0,30) || 'site';
  a.href = URL.createObjectURL(blob);
  a.download = fname + '.html';
  document.body.appendChild(a); a.click(); a.remove();
  toast('Сайт опубликован! Файл скачан.','success');
}


function exportJson(){
  const blob = new Blob([JSON.stringify(DATA,null,2)], {type:'application/json'});
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = 'site-backup-' + new Date().toISOString().slice(0,10) + '.json';
  document.body.appendChild(a); a.click(); a.remove();
  toast('Данные экспортированы','success');
}
function importJson(e){
  const f = e.target.files[0]; if(!f) return;
  const r = new FileReader();
  r.onload = ev=>{
    try{
      const obj = JSON.parse(ev.target.result);
      if(!obj.settings || !Array.isArray(obj.cards)) throw new Error('bad');
      DATA = obj;
      DATA.categories = DATA.categories||[]; DATA.orders = DATA.orders||[];
      DATA.settings.customFields = DATA.settings.customFields||[];
      saveData(); applyTheme(); adminData();
      toast('Данные импортированы','success');
    }catch(err){ toast('Ошибка: неверный файл','error'); }
  };
  r.readAsText(f);
  e.target.value='';
}
